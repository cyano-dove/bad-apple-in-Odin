Understanding the .mp4 data

We will start by taking the .mp4 and print out in hex data

As you can clearly see, I don’t know what I am doing here. So let’s look at some resource.

https://dev.to/alfg/a-quick-dive-into-mp4-57fo

basically every .mp4 file has boxes (or atoms) stored in a container. There are boxes for video, audio, subtitle, etc. The first 8 bytes of a box are designated as the box header, 4 for the size of the box and 4 for the name of the box, from there you can figure that the data size is the box size - 8.

├───mdat -------------------> Movie Data Box
├───moov -------------------> Movie Boxes
│   ├───trak ---------------> Track Box
│   │   ├─── tkhd ----------> Track Header
│   │   └─── mdia ----------> Media Box
│   │        └─── ...
│   └───trak
│   │   ├─── tkhd ----------> Track Header
│   │   └─── mdia ----------> Media Box
│   │        └─── ...
└───udta -------------------> Userdata Box

When one of the frames gets put into a hex viewer, the first few bytes are the most notable, P6 is the “magic number” identifying the file type, 480 is the width of the image and 360 is the height of the image and 255 is the maximum color value, whatever that is.

Every .ppm file has a header like this with 3 newline, so it’s easy to gather width, height and offset the header out off the pixels data

frameHeader :: proc(data: []u8, print := false) -> (offset: int, width: int, height: int) {
    newLineCount : int = 0
    offset : int
    frameW : int
    frameH : int
    for value, index in data {
        if value == 10 {
            newLineCount = newLineCount + 1
        }
        if newLineCount == 1 && frameW == 0 {
            tempIndex := 1
            for ; data[tempIndex+index] != u8(' '); tempIndex += 1 {
                frameW = frameW * 10 + int(data[tempIndex+index]) - '0'
            }
            tempIndex += 1
            for ; data[tempIndex+index] != u8('\n'); tempIndex += 1 {
                frameH = frameH * 10 + int(data[tempIndex+index]) - '0'
            }
        }
        if newLineCount == 3 {
            offset = index
            break
        }
    }
    if print {
        fmt.printf("offset: %d | width: %d | height: %d", offset, frameW, frameH)
    }
    return offset, frameW, frameH
}

framePixelData :: proc(data: []u8) -> (pixels: []u8, w: int, h: int) {
    offset, width, height := frameHeader(data)
    r, g, b : u8
    pixel := make([]u8, width * height) // avg the r, g, b value into pixel's brightness

    for y in 0..<height {
        for x in 0..<width {
            p := y * width + x
            index := offset + 1 + p*3
            r = data[index    ]
            g = data[index + 1]
            b = data[index + 2]
            if (int(r) + int(g) + int(b)) / 3 > 128{
                pixel[p] = 255
            }
        }
    }
    return pixel, width, height
}

printFrame :: proc(pixel: []u8, width, height : int, setScrWidth := -1, setScrHeight := -1) {
    scaleWidth : f32 = 1
    scaleHeight : f32 = 1
    scrWidth := width
    scrHeight := height
    if setScrWidth != -1 {
        scrWidth = setScrWidth
        scaleWidth = f32(width) / f32(scrWidth)
        if scaleWidth < 1 {
            scaleWidth = 1
        }
    }
    if setScrHeight != -1 {
        scrHeight = setScrHeight
        scaleHeight = f32(height) / f32(scrHeight)
        if scaleHeight < 1 {
            scaleHeight = 1
        }
    }

    str := strings.builder_make()
    defer strings.builder_destroy(&str)

    if scaleHeight == 1 && scaleWidth == 1 {
        // without scaling
        for y in 0..<height {
            for x in 0..<width {
                if pixel[y * width + x] == 255 {
                    strings.write_byte(&str, '#')
                }
                else {
                    strings.write_byte(&str, '.')
                }
            }
            strings.write_byte(&str, '\n')
        }
    }
    else{
        // with scaling
        BRIGHTNESS := " .`',:;-~=+*#%$@" // shoutout to the C donut devblog https://www.a1k0n.net/2021/01/13/optimizing-donut.html
        for scrY in 0..<scrHeight {
            for scrX in 0..<scrWidth {
                sum := 0
                count := 0
                for y in scrY*int(scaleHeight) ..< (scrY+1)*int(scaleHeight) {
                    for x in scrX*int(scaleWidth) ..< (scrX+1)*int(scaleWidth) {
                        sum += int(pixel[y*width + x])
                        count += 1
                    }
                }
                strings.write_byte(&str, BRIGHTNESS[int(sum/count) / 17])
            }
            strings.write_byte(&str, '\n')
        }
    }
    os.write_string(os.stdout, strings.to_string(str))
    defer delete(pixel)
}

iterateFrames :: proc(folder : string, width : int, height : int) {
    for n := 1; ; n += 1 {
        currTime := time.tick_now()

        frame := fmt.tprintf("%s/%04d.ppm", folder, n)
        data, err := os.read_entire_file(frame, context.allocator)

            break
        }

        delete(data)
        free_all(context.temp_allocator)

        os.write_string(os.stdout, HOME)
        diffTime := time.tick_since(currTime)
        if diffTime < FRAME_TIME {
            time.sleep(FRAME_TIME - diffTime)
        }
    }
}

This works, but when I play this video it will lag behind, might be a Windows thing. After a bit of looking around the problem seems to be the loop measuring being off. To resolve this, I put the time.tick_now() outside of the loop.

iterateFrames :: proc(folder : string, width : int, height : int) {
    startTime := time.tick_now()
    for n := 1; ; n += 1 {

        frame := fmt.tprintf("%s/%04d.ppm", folder, n)
        data, err := os.read_entire_file(frame, context.allocator)
        if err != nil {
            break
        }
        printFrame(framePixelData(data), width, height)
        delete(data)
        free_all(context.temp_allocator)
        os.write_string(os.stdout, HOME)

        nextFrame  := time.Duration(n) * FRAME_TIME
        elapsed := time.tick_since(startTime)
        if elapsed < nextFrame {
            time.sleep(nextFrame - elapsed)
        }
    }
}

printFrame :: proc(pixel: []u8, width, height : int, setScrWidth := -1, setScrHeight := -1) {
    defer delete(pixel)
    scaleWidth : f32 = 1
    scaleHeight : f32 = 1
    scrWidth := width
    scrHeight := height
    if setScrWidth != -1 {
        scrWidth = min(setScrWidth, width)
        scaleWidth = f32(width) / f32(scrWidth)
    }
    if setScrHeight != -1 {
        scrHeight = min(setScrHeight, height)
        scaleHeight = f32(height) / f32(scrHeight)
    }
    
    str := strings.builder_make()
    defer strings.builder_destroy(&str)

    if scaleHeight == 1 && scaleWidth == 1 {
        // without scaling
        for y in 0..<height {
            for x in 0..<width {
                if pixel[y * width + x] == 255 {
                    strings.write_byte(&str, '#')
                }
                else {
                    strings.write_byte(&str, '.')
                }
            }
            strings.write_byte(&str, '\n') 
        }
    }
    else{
        // with scaling
        BRIGHTNESS := " .`',:;-~=+*#%$@" // shoutout to the C donut devblog https://www.a1k0n.net/2021/01/13/optimizing-donut.html
        for scrY in 0..<scrHeight {
            yStart := int(f32(scrY)     * scaleHeight)
            yEnd := int(f32(scrY + 1) * scaleHeight)
            for scrX in 0..<scrWidth {
                xStart := int(f32(scrX)     * scaleWidth)
                xEnd := int(f32(scrX + 1) * scaleWidth)
                sum := 0
                count := 0
                for y in yStart..<yEnd {
                    for x in xStart..<xEnd {
                        sum += int(pixel[y*width + x])
                        count += 1
                    }
                }
                strings.write_byte(&str, BRIGHTNESS[int(sum/count) / 17])
            }
            strings.write_byte(&str, '\n')  
        }
    }
    os.write_string(os.stdout, strings.to_string(str))
}

iterateFrames :: proc(folder : string, width : int, height : int) {
    engine : aud.engine
    result := aud.engine_init(nil, &engine)
    if result != .SUCCESS {
        fmt.println("could not start audio, press enter to continue: ", result)
        libc.getchar()
        return
    }
    defer aud.engine_uninit(&engine)
    aud.engine_play_sound(&engine, fmt.ctprintf("%s/audio.wav", folder), nil)
    
    os.write_string(os.stdout, ALT_SCR_T + CUR_HIDE + CLEAR)
    defer os.write_string(os.stdout, CUR_SHOW + ALT_SCR_F)

    startTime := time.tick_now()
    for n := 1; ; n += 1 {

        frame := fmt.tprintf("%s/%04d.ppm", folder, n)
        data, err := os.read_entire_file(frame, context.allocator)
        if err != nil {
            break
        }
        printFrame(framePixelData(data), width, height)
        delete(data)
        free_all(context.temp_allocator)
        os.write_string(os.stdout, HOME)

        nextFrame  := time.Duration(n) * FRAME_TIME
        elapsed := time.tick_since(startTime)
        if elapsed < nextFrame {
            time.sleep(nextFrame - elapsed)
        }
    }
}

main :: proc() {
    if len(os.args) < 2 {
        fmt.println("args: \"file\" width(optional) height(optional)")
        return
    }
    file := os.args[1]
    dot := strings.last_index_byte(file, '.')
    if dot == -1 {
        fmt.println("file needs an extension, e.g. video.mp4")
        return
    }
    folder := fmt.aprintf("%s_frames", file[:dot])
    defer delete(folder)
    width : int = -1
    height : int = -1
    if len(os.args) > 2 {
        width, _ = strconv.parse_int(os.args[2])
    }
    if len(os.args) > 3 {
        height, _ = strconv.parse_int(os.args[3])
    }

    makeFramesFF(file, folder)
    fmt.println("press enter to play")
    libc.getchar()


    iterateFrames(folder, width, height)

    libc.getchar()
}
